[binaries]
# Meson 0.53.2 doesn't use any cflags when doing basic compiler tests,
# so we have to add -nostdlib to the compiler configuration itself or
# early compiler tests will fail. This can be removed when picolibc
# requires at least version 0.54.2 of meson.
c = ['mos-unknown-unknown-clang', '-nostdlib', '-flto', '-D__COMPILING_MOS_LIBC__',
     '-D__IEEE_LITTLE_ENDIAN', '-D_LDBL_EQ_DBL', '-Oz', '-g',
     '-Wno-tautological-compare' ]
cpp = ['mos-unknown-unknown-clang++', '-nostdlib', '-flto', '-D__COMPILING_MOS_LIBC__',
     '-D__IEEE_LITTLE_ENDIAN', '-D_LDBL_EQ_DBL', '-Oz', '-g',
     '-Wno-tautological-compare' ]
ar = 'llvm-ar'
as = 'mos-unknown-unknown-clang'
nm = 'llvm-nm'
strip = 'llvm-strip'

[host_machine]
system = 'none'
cpu_family = 'mos'
cpu = 'mos'
endian = 'little'

[properties]
skip_sanity_check = true
exe_wrapper = ['sh', '-c', 'test -z "$PICOLIBC_TEST" || run-zbc "$@"', 'run-zbc']

# ld.lld supports --defsym but meson can't detect it during configure
# because there's no libc yet. Override the detection.
has_link_defsym = true

# llvm-mos runtime library (provides __mulsi3, __divsi3, etc.)
# The libgcc property is set by do-mos-configure via a supplementary cross file
# using mos-clang --print-runtime-dir to find the library path dynamically

# ZBC memory layout for MOS 6502:
# RAM: 0x0200 - 0xFCDF (before ZBC device at 0xFCE0)
# Stack grows down from end of RAM
default_flash_addr = '0x0200'
default_flash_size = '0xFAE0'
default_ram_addr = '0x0200'
default_ram_size = '0xFAE0'

# llvm-mos imaginary registers and linker script extensions
# These are loaded from separate files for readability
linker_preamble = 'INCLUDE picocrt/machine/mos/mos-regs.ld'
extra_memory = 'INCLUDE picocrt/machine/mos/mos-memory.ld'
extra_sections = 'INCLUDE picocrt/machine/mos/mos-sections.ld'

# Build each math function as a separate test executable.
# Required for MOS 6502 due to 64KB memory limit.
per_function_math_tests = true

# Use minimal error output in math tests to reduce code size.
# Error messages only show function:line instead of full debug values.
tiny_math_test = true
