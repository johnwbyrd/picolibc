[binaries]
# Meson 0.53.2 doesn't use any cflags when doing basic compiler tests,
# so we have to add -nostdlib to the compiler configuration itself or
# early compiler tests will fail. This can be removed when picolibc
# requires at least version 0.54.2 of meson.
c = ['mos-unknown-unknown-clang', '-nostdlib', '-flto', '-D__COMPILING_MOS_LIBC__',
     '-D__IEEE_LITTLE_ENDIAN', '-D_LDBL_EQ_DBL', '-Oz', '-g',
     '-Wno-tautological-compare' ]
cpp = ['mos-unknown-unknown-clang++', '-nostdlib', '-flto', '-D__COMPILING_MOS_LIBC__',
     '-D__IEEE_LITTLE_ENDIAN', '-D_LDBL_EQ_DBL', '-Oz', '-g',
     '-Wno-tautological-compare' ]
ar = 'llvm-ar'
as = 'mos-unknown-unknown-clang'
nm = 'llvm-nm'
strip = 'llvm-strip'
# exe_wrapper must be in [binaries] section, not [properties]
# Finds run-zbc by walking up from the ELF path to find the source tree
exe_wrapper = ['sh', '-c', 'test -z "$PICOLIBC_TEST" || { d="$1"; while [ ! -f "$d/scripts/run-zbc" ] && [ "$d" != "/" ]; do d="$(dirname "$d")"; done; "$d/scripts/run-zbc" "$@"; }', 'run-zbc']

[host_machine]
system = 'none'
cpu_family = 'mos'
cpu = 'mos'
endian = 'little'

[properties]
skip_sanity_check = true

# ld.lld supports --defsym but meson can't detect it during configure
# because there's no libc yet. Override the detection.
has_link_defsym = true

# llvm-mos runtime library (provides __mulsi3, __divsi3, etc.)
# The libgcc property is set by do-mos-configure via a supplementary cross file
# using mos-clang --print-runtime-dir to find the library path dynamically

# ZBC memory layout for MOS 6502:
# RAM: 0x0200 - 0xFCDF (before ZBC device at 0xFCE0)
# Stack grows down from end of RAM
default_flash_addr = '0x0200'
default_flash_size = '0xFAE0'
default_ram_addr = '0x0200'
default_ram_size = '0xFAE0'

# llvm-mos imaginary registers and linker script extensions
# These are loaded from separate files for readability
linker_preamble = 'INCLUDE picocrt/machine/mos/mos-regs.ld'
extra_memory = 'INCLUDE picocrt/machine/mos/mos-memory.ld'
extra_sections = 'INCLUDE picocrt/machine/mos/mos-sections.ld'

# Only build tests that fit within 64KB memory limit AND complete in reasonable time.
# Tests not in this list will be skipped during build.
# Note: memcpy-1 and strcmp-1 removed - they have millions of iterations,
# taking hours on emulated 1MHz 6502.
allowed_tests = ['math_test_acos', 'math_test_acosf', 'math_test_acoshf', 'math_test_asin', 'math_test_asinf', 'math_test_asinhf', 'math_test_atan', 'math_test_atanf', 'math_test_atanh', 'math_test_atanhf', 'math_test_ceil', 'math_test_ceilf', 'math_test_copysign', 'math_test_copysignf', 'math_test_cosf', 'math_test_cosh', 'math_test_coshf', 'math_test_erf', 'math_test_erfc', 'math_test_erfcf', 'math_test_erff', 'math_test_exp', 'math_test_expf', 'math_test_fabs', 'math_test_fabsf', 'math_test_floor', 'math_test_floorf', 'math_test_fmod', 'math_test_fmodf', 'math_test_hypot', 'math_test_hypotf', 'math_test_issignaling', 'math_test_log', 'math_test_log10', 'math_test_log10f', 'math_test_log1p', 'math_test_log1pf', 'math_test_log2', 'math_test_log2f', 'math_test_logf', 'math_test_modf', 'math_test_modff', 'math_test_scalb', 'math_test_scalbn', 'math_test_sinf', 'math_test_sinh', 'math_test_sinhf', 'math_test_sqrt', 'math_test_sqrtf', 'math_test_tanf', 'math_test_tanh', 'math_test_tanhf', 'math_test_trunc', 'math_test_truncf', 'size_max', 'memmove1', 'tstring', 'tiswctype', 'twctrans']
