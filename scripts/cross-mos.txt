[binaries]
# Meson 0.53.2 doesn't use any cflags when doing basic compiler tests,
# so we have to add -nostdlib to the compiler configuration itself or
# early compiler tests will fail. This can be removed when picolibc
# requires at least version 0.54.2 of meson.
c = ['mos-clang', '-nostdlib', '-flto', '-D__COMPILING_MOS_LIBC__',
     '-D__IEEE_LITTLE_ENDIAN', '-D_LDBL_EQ_DBL', '-Os',
     '-Wno-tautological-compare' ]
cpp = ['mos-clang', '-nostdlib', '-flto', '-D__COMPILING_MOS_LIBC__',
     '-D__IEEE_LITTLE_ENDIAN', '-D_LDBL_EQ_DBL', '-Os',
     '-Wno-tautological-compare' ]
ar = 'llvm-ar'
as = 'mos-clang'
nm = 'llvm-nm'
strip = 'llvm-strip'

[host_machine]
system = 'none'
cpu_family = 'mos'
cpu = 'mos'
endian = 'little'

[properties]
skip_sanity_check = true
exe_wrapper = ['sh', '-c', 'test -z "$PICOLIBC_TEST" || run-zbc "$@"', 'run-zbc']

# ld.lld supports --defsym but meson can't detect it during configure
# because there's no libc yet. Override the detection.
has_link_defsym = true

# llvm-mos runtime library (provides __mulsi3, __divsi3, etc.)
# The libgcc property is set by do-mos-configure via a supplementary cross file
# using mos-clang --print-runtime-dir to find the library path dynamically

# ZBC memory layout for MOS 6502:
# RAM: 0x0200 - 0xFCDF (before ZBC device at 0xFCE0)
# Stack grows down from end of RAM
default_flash_addr = '0x0200'
default_flash_size = '0xFAE0'
default_ram_addr = '0x0200'
default_ram_size = '0xFAE0'

# llvm-mos imaginary registers (zero page 0x00-0x1F)
# These are required for code generation on 6502
# Note: Must be single line - meson cross files don't support multiline strings
linker_preamble = '/* llvm-mos imaginary registers */ __rc0 = 0x00; __rc1 = __rc0 + 1; PROVIDE(__rc2 = __rc1 + 1); __rc3 = __rc2 + 1; PROVIDE(__rc4 = __rc3 + 1); __rc5 = __rc4 + 1; PROVIDE(__rc6 = __rc5 + 1); __rc7 = __rc6 + 1; PROVIDE(__rc8 = __rc7 + 1); __rc9 = __rc8 + 1; PROVIDE(__rc10 = __rc9 + 1); __rc11 = __rc10 + 1; PROVIDE(__rc12 = __rc11 + 1); __rc13 = __rc12 + 1; PROVIDE(__rc14 = __rc13 + 1); __rc15 = __rc14 + 1; PROVIDE(__rc16 = __rc15 + 1); __rc17 = __rc16 + 1; PROVIDE(__rc18 = __rc17 + 1); __rc19 = __rc18 + 1; PROVIDE(__rc20 = __rc19 + 1); __rc21 = __rc20 + 1; PROVIDE(__rc22 = __rc21 + 1); __rc23 = __rc22 + 1; PROVIDE(__rc24 = __rc23 + 1); __rc25 = __rc24 + 1; PROVIDE(__rc26 = __rc25 + 1); __rc27 = __rc26 + 1; PROVIDE(__rc28 = __rc27 + 1); __rc29 = __rc28 + 1; PROVIDE(__rc30 = __rc29 + 1); __rc31 = __rc30 + 1; PROVIDE(__stack = 0xFCDF); PROVIDE(__call_indir = 0);'
